ORG 0x14;
A1: WORD 0;
A2: WORD 1;
A3: WORD 0;
A4: WORD 1;
LEN: WORD 3;
CURRINDX: WORD 0x14;

START:CLA;
ADD #0x14;
ADD LEN;
ST $CURRINDX;
CALL $SHOW;

CYCLE: CLA;
ADD #0x14;
ADD LEN;
ST $CURRINDX;
CALL MINUS;
BEQ GETOUT;
LD $CURRINDX;
DEC;
ST $CURRINDX;
CLA;
CALL $SHOW;
JUMP CYCLE;
GETOUT: HLT;


ORG 0x2F;
MINUS: LD (CURRINDX);
DEC;
BMI NEGAT;
ST (CURRINDX)+;
LD #1;
RET;
NEGAT: LD $CURRINDX;
SUB #0x14;
BEQ RETURN;
DEC;
ADD #0x14;
ST $CURRINDX;
CALL MINUS;
SUB #0x1;
BNE RETURN;
LD #0x9;
ST (CURRINDX)+;
LD #1;
RET;
RETURN: CLA;
RET;






ORG 0x5F;
SHOW: LD (CURRINDX);
PUSH;
LD $CURRINDX;
SUB #0x14;
SUB $LEN;
NEG;
PUSH;
CALL $OUTPUT;
POP;
POP;
LD $CURRINDX;
DEC;
ST $CURRINDX;
SUB #0x14;
BMI DONE;
CALL $SHOW;
DONE: RET;



ORG 0x100;
OUTPUT: LD &1;
ASL;
ASL;
ASL;
ASL;
ADD &2;
OUT 0x14;
RET;
